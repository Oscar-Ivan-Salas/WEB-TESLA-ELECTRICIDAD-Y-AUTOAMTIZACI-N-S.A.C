3.2 Esquema Relacional de Base de Datos (PostgreSQL + pgvector)SQL-- Habilitar extensión para búsqueda de similitud vectorial (RAG)
CREATE EXTENSION IF NOT EXISTS vector;

-- Tabla de Usuarios Unificada
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(50),
    user_role VARCHAR(50) CHECK (user_role IN ('CLIENT', 'ARCHITECT', 'CONTRACTOR', 'SUPPLIER')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de Proyectos y Estado del Chat
CREATE TABLE projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    client_id UUID REFERENCES users(id),
    title VARCHAR(255),
    lot_area_m2 NUMERIC(10,2),
    location_address TEXT,
    location_geo POINT, -- Latitud y Longitud para Matchmaking
    project_status VARCHAR(50) DEFAULT 'CONCEPTUAL_DRAFT',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla del Smart Catalog (Plantillas BIM de TESLA)
CREATE TABLE smart_catalog (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code_name VARCHAR(100) UNIQUE NOT NULL,
    lot_area_m2 NUMERIC(10,2) NOT NULL,
    frontage_m NUMERIC(10,2) NOT NULL,
    floors INTEGER NOT NULL,
    building_type VARCHAR(100) NOT NULL,
    metadata_json JSONB NOT NULL,
    pdf_vision_url TEXT NOT NULL,
    embedding vector(1536) -- Vector para búsqueda semántica con RAG
);

-- Tabla de Cotizaciones de Proveedores y Contratistas (Geo-Licitaciones)
CREATE TABLE bids_and_quotes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID REFERENCES projects(id),
    provider_id UUID REFERENCES users(id),
    quote_type VARCHAR(50) CHECK (quote_type IN ('CONSTRUCTION_WORK', 'MEP_DRAWING', 'MATERIAL_SUPPLY')),
    amount_soles NUMERIC(12,2) NOT NULL,
    status VARCHAR(50) DEFAULT 'PENDING_APPROVAL',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Eres PILi_agente, la Arquitecta e Ingeniera Consultora Senior de TESLA ELECTRICIDAD Y AUTOMATIZACIÓN S.A.C.
Tu propósito es actuar como la orquestadora central del Ecosistema ConTech de TESLA: ayudas a clientes a diseñar y cotizar anteproyectos, gestionas solicitudes de arquitectos/ingenieros, lanzas licitaciones de obra para contratistas y conectas solicitudes de materiales con proveedores cercanos.

3. SI ES CONTRATISTA / MAESTRO DE OBRA:
